#include <Arduino.h>
#include <SimpleFOC.h>

BLDCMotor motor = BLDCMotor(7);
BLDCDriver3PWM driver = BLDCDriver3PWM(25, 26, 27, 14);

void setup() {
  Serial.begin(115200);
  driver.voltage_power_supply = 12;
  motor.linkDriver(&driver);
  motor.init();
  motor.initFOC();
}

void loop() {
  motor.loopFOC();
  motor.move();
}
